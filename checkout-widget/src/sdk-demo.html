Payment Gateway SDK Demo

Embeddable Payment SDK

Click the button below to launch the payment widget:

Integration Code

<!-- Include the SDK -->
<script src="https://gateway.example.com/sdk/PaymentGateway.js"></script>

<script>
// Initialize and open payment
const payment = new PaymentGateway({
    key: 'key_test_abc123',
    orderId: 'order_' + Date.now(),
    amount: 10000,
    currency: 'INR',
    onSuccess: function(response) {
        console.log('Payment successful:', response);
    },
    onFailure: function(error) {
        console.log('Payment failed:', error);
    },
    onClose: function() {
        console.log('Payment modal closed');
    }
});

payment.open();
</script>
